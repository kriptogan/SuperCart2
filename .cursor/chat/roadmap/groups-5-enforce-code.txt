Update BurgerMenu composable:
---
@Composable
fun BurgerMenu(
    onManageCategoriesClick: () -> Unit = {},
    onManageStoresClick: () -> Unit = {},
    onManageGroupClick: () -> Unit = {},
    onUploadClick: () -> Unit = {},
    onDownloadClick: () -> Unit = {}
) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    var expanded by remember { mutableStateOf(false) }
    var hasGroupCode by remember { mutableStateOf(false) }
    
    // Check if group code exists
    LaunchedEffect(Unit) {
        hasGroupCode = DataStoreManager.hasGroupCode(context)
    }
    
    // ... existing code ...
    
    DropdownMenu(
        expanded = expanded,
        onDismissRequest = { expanded = false }
    ) {
        // ... existing menu items ...
        
        Divider()
        
        // Upload to Cloud
        DropdownMenuItem(
            text = {
                Column {
                    Text("Upload to Cloud")
                    if (!hasGroupCode) {
                        Text(
                            "Requires family group",
                            fontSize = 10.sp,
                            color = Color.Red
                        )
                    }
                }
            },
            onClick = {
                if (hasGroupCode) {
                    expanded = false
                    onUploadClick()
                } else {
                    Toast.makeText(
                        context,
                        "Please create or join a family group first",
                        Toast.LENGTH_LONG
                    ).show()
                }
            },
            enabled = hasGroupCode  // Disable if no code
        )
        
        // Download from Cloud
        DropdownMenuItem(
            text = {
                Column {
                    Text("Download from Cloud")
                    if (!hasGroupCode) {
                        Text(
                            "Requires family group",
                            fontSize = 10.sp,
                            color = Color.Red
                        )
                    }
                }
            },
            onClick = {
                if (hasGroupCode) {
                    expanded = false
                    onDownloadClick()
                } else {
                    Toast.makeText(
                        context,
                        "Please create or join a family group first",
                        Toast.LENGTH_LONG
                    ).show()
                }
            },
            enabled = hasGroupCode  // Disable if no code
        )
    }
}

📐 KEY DESIGN DECISIONS
═══════════════════════════════════════════════════════════════════

1. **UI Approach:**
   - Disable buttons (grayed out)
   - Show "Requires family group" hint in red
   - Toast message if clicked when disabled

2. **Check Timing:**
   - LaunchedEffect when menu opens
   - Fresh check each time

3. **User Guidance:**
   - Clear message about what's needed
   - Direct user to "Family Group" menu item

4. **Backend Safety:**
   - FirebaseManager already checks for code
   - UI is first line of defense
   - Backend still validates

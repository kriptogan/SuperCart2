🏗️ DELETE CONFIRMATION LOGIC
═══════════════════════════════════════════════════════════════════

Delete Confirmation Dialog:
---
if (showDeleteConfirmation != null) {
    AlertDialog(
        onDismissRequest = { showDeleteConfirmation = null },
        title = { Text("Delete Store") },
        text = { 
            Text("Are you sure you want to delete ${showDeleteConfirmation?.name}? " +
                 "This will unlink all groceries from this store.") 
        },
        confirmButton = {
            TextButton(
                onClick = {
                    scope.launch {
                        showDeleteConfirmation?.let { store ->
                            DataManagerObject.deleteStore(store.uuid)
                            DataStoreManager.saveDataGlobally()
                        }
                        showDeleteConfirmation = null
                    }
                }
            ) {
                Text("Delete", color = Color.Red)
            }
        },
        dismissButton = {
            TextButton(onClick = { showDeleteConfirmation = null }) {
                Text("Cancel")
            }
        }
    )
}
---

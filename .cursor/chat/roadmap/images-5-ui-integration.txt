Add Image Display Section (at top of dialog content):

// Image Section (at the top of the dialog content Column)
Card(
    modifier = Modifier
        .fillMaxWidth()
        .height(200.dp)
        .clickable { showImageSourceDialog = true },
    colors = CardDefaults.cardColors(
        containerColor = SuperCartColors.lightGray.copy(alpha = 0.3f)
    ),
    shape = SuperCartShapes.medium
) {
    Box(
        modifier = Modifier.fillMaxSize(),
        contentAlignment = Alignment.Center
    ) {
        if (isProcessingImage) {
            // Show loading indicator
            Column(
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                CircularProgressIndicator(
                    color = SuperCartColors.primaryGreen
                )
                Spacer(modifier = Modifier.height(8.dp))
                Text(
                    text = "Processing image...",
                    color = SuperCartColors.gray,
                    fontSize = 12.sp
                )
            }
        } else if (currentImageUUID != null) {
            // Show image using Coil
            val imageFile = remember(currentImageUUID) {
                ImageManager.getLocalImageFile(currentImageUUID!!, context)
            }
            
            if (imageFile != null && imageFile.exists()) {
                AsyncImage(
                    model = imageFile,
                    contentDescription = "Grocery Image",
                    modifier = Modifier.fillMaxSize(),
                    contentScale = ContentScale.Crop
                )
                
                // Remove button overlay (top-right)
                IconButton(
                    onClick = {
                        // Remove image
                        currentImageUUID?.let { uuid ->
                            ImageManager.deleteLocalImage(uuid, context)
                        }
                        currentImageUUID = null
                        imageUri = null
                    },
                    modifier = Modifier
                        .align(Alignment.TopEnd)
                        .padding(8.dp)
                ) {
                    Icon(
                        imageVector = Icons.Default.Close,
                        contentDescription = "Remove Image",
                        tint = Color.White,
                        modifier = Modifier
                            .background(
                                Color.Black.copy(alpha = 0.5f),
                                shape = CircleShape
                            )
                            .padding(4.dp)
                    )
                }
            } else {
                // Image file not found
                PlaceholderContent()
            }
        } else {
            // No image - show placeholder
            PlaceholderContent()
        }
    }
}

@Composable
private fun PlaceholderContent() {
    Column(
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Icon(
            imageVector = Icons.Default.AddPhotoAlternate,

            modifier = Modifier.size(48.dp)
        )
        Spacer(modifier = Modifier.height(8.dp))
        Text(
            text = "Add Image",
            color = SuperCartColors.gray,
            fontSize = 14.sp
        )
        Text(
            text = "Tap to upload",
            color = SuperCartColors.gray,
            fontSize = 12.sp
        )
    }
}

Add Image Processing Logic:
---
// ImageSourceDialog
if (showImageSourceDialog) {
    ImageSourceDialog(
        onDismiss = { showImageSourceDialog = false },
        onImageSelected = { uri ->
            // Process image in background
            scope.launch {
                isProcessingImage = true
                
                try {
                    // Generate new UUID for image if creating new, or use existing
                    val imageUUID = currentImageUUID ?: UUID.randomUUID().toString()
                    
                    // Compress and save locally
                    val savedFile = ImageManager.compressAndSaveImage(uri, imageUUID, context)
                    
                    if (savedFile != null) {
                        currentImageUUID = imageUUID
                        imageUri = Uri.fromFile(savedFile)
                        android.util.Log.d("GroceryCreationDialog", "Image saved: $imageUUID")
                    } else {
                        android.util.Log.e("GroceryCreationDialog", "Failed to save image")
                        // Show error to user (optional)
                    }
                } catch (e: Exception) {
                    android.util.Log.e("GroceryCreationDialog", "Error processing image", e)
                } finally {
                    isProcessingImage = false
                }
            }
        }
    )
}

📐 IMAGE PROCESSING FLOW
═══════════════════════════════════════════════════════════════════

When User Selects/Captures Image:
---
1. Receive Uri from picker/camera
2. Set isProcessingImage = true (show loading)
3. Launch coroutine:
   a. Generate UUID (new or existing)
   b. Call ImageManager.compressAndSaveImage()
   c. If successful: Update currentImageUUID
   d. If failed: Log error, show message
4. Set isProcessingImage = false
5. Image appears in preview
---

When User Removes Image:
---
1. Delete local file (ImageManager.deleteLocalImage)
2. Set currentImageUUID = null
3. Set imageUri = null
4. Placeholder appears
5. When saving grocery: imageUUID will be null
---


⚠️ IMPORTANT CONSIDERATIONS
═══════════════════════════════════════════════════════════════════

1. **Image UUID Management:**
   - Create mode: Generate new UUID when image selected
   - Edit mode: Use existing UUID, or generate if adding first image
   - Preserve UUID when updating grocery (don't regenerate)

2. **Image Deletion:**
   - When removing image from grocery: Delete local file immediately
   - When deleting grocery: Delete image in DataManagerObject.deleteGrocery()

3. **Memory Management:**
   - Coil handles bitmap recycling
   - Don't hold large bitmaps in memory
   - Use File references, not Bitmap state

4. **Loading States:**
   - Show progress during compression (can take 1-3 seconds for large images)
   - Disable save button while processing
   - Clear visual feedback
